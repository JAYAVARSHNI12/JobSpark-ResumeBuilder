pipeline {
    agent any

    environment {
        // ==============================================
        // UPDATE THESE VALUES FOR YOUR OWN REGISTRY/SERVER
        // ==============================================
        REGISTRY      = 'docker.io/jayavarshni'   // Change this!
        IMAGE_NAME    = 'jobspark-resume-builder'
        IMAGE_TAG     = "${env.BUILD_NUMBER}"
        DEPLOY_SERVER = 'your-server-ip'           // Optional: change or remove
        DEPLOY_USER   = 'deployer'                 // Optional
        DEPLOY_PORT   = '8080'
    }

    stages {
        stage('Checkout') {
            steps {
                checkout scm
            }
        }

        stage('Build Docker Image') {
            steps {
                script {
                    docker.build("${REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}", ".")
                }
            }
        }

        stage('Push Docker Image') {
            steps {
                script {
                    docker.withRegistry("https://${REGISTRY}", 'docker-credentials-id') {
                        docker.image("${REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}").push()
                        docker.image("${REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}").push('latest')
                    }
                }
            }
        }

    }

    post {
        always {
            cleanWs()
        }
        success {
            echo "Pipeline succeeded! Image: ${REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}"
        }
        failure {
            echo "Pipeline failed."
        }
    }
}
